-- Usar [py] o [python] segun como este instalado Python
-- Agregar -q al final para correr sin ventana (solo texto, rapido)

-- 1. Laberinto chico (ideal para explicar), los 4 algoritmos
py pacman.py -l tp2Maze -p SearchAgent -a fn=dfs
py pacman.py -l tp2Maze -p SearchAgent -a fn=bfs
py pacman.py -l tp2Maze -p SearchAgent -a fn=ucs
py pacman.py -l tp2Maze -p SearchAgent -a fn=astar,heuristic=manhattanHeuristic

-- 2. Laberinto mediano, los 4 algoritmos
py pacman.py -l mediumMaze -p SearchAgent -a fn=dfs
py pacman.py -l mediumMaze -p SearchAgent -a fn=bfs
py pacman.py -l mediumMaze -p SearchAgent -a fn=ucs
py pacman.py -l mediumMaze -p SearchAgent -a fn=astar,heuristic=manhattanHeuristic

-- 3. Costo uniforme con costos distintos por casillero
--    (StayEast: el lado derecho es barato / StayWest: el lado izquierdo es barato)
py pacman.py -l mediumDottedMaze -p StayEastSearchAgent
py pacman.py -l mediumScaryMaze -p StayWestSearchAgent

-- 4. Laberinto abierto: A* expande menos nodos que BFS/UCS
py pacman.py -l openMaze -p SearchAgent -a fn=bfs
py pacman.py -l openMaze -p SearchAgent -a fn=astar,heuristic=manhattanHeuristic
